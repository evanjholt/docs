---
title: "Infrastructure Projects"
api: "GET /api/v1/infrastructure_projects"
description: "Retrieve a list of infrastructure projects funded by the Canadian government."
---

Get a standardized list of federal infrastructure funding projects, filterable by region, timeline, and project type.

### Query Parameters

<ParamField query="q" type="string">
  Full-text match across title, recipient, and location
</ParamField>

<ParamField query="province" type="string">
  Two-letter province code
</ParamField>

<ParamField query="program" type="string">
  Partial match on `program_en`
</ParamField>

<ParamField query="status" type="string">
  Exact match on project status
</ParamField>

<ParamField query="min_value" type="number">
  Minimum federal contribution
</ParamField>

<ParamField query="max_value" type="number">
  Maximum federal contribution
</ParamField>

<ParamField query="issued_after" type="string">
  ISO 8601 approved-date lower bound
</ParamField>

<ParamField query="issued_before" type="string">
  ISO 8601 approved-date upper bound
</ParamField>

<ParamField query="sort_by" type="string">
  `value` or `date`. Default: `date`
</ParamField>

<ParamField query="sort_order" type="string">
  `asc` or `desc`. Default: `desc`
</ParamField>

<ParamField query="limit" type="integer" default="100">
  Result limit. Default: `100`. Max: `200`
</ParamField>

<ParamField query="offset" type="integer" default="0">
  Result offset. Default: `0`
</ParamField>

```python Python
}

print(response.json())
```
