---
title: "MPs & Senators"
api: "GET /api/v1/politicians"
description: "Retrieve a list of Members of Parliament and Senators."
---

Get a standardized list of active and historical federal politicians, their party affiliations, ridings, and roles.

### Query Parameters

<ParamField query="q" type="string">
  Partial match on politician name.
</ParamField>

<ParamField query="party" type="string">
  Partial match on political party.
</ParamField>

<ParamField query="province" type="string">
  Filter by province or territory represented.
</ParamField>

<ParamField query="current" type="boolean">
  Defaults to `true`; set `false` to include historical politicians.
</ParamField>

<ParamField query="page" type="integer">
  Page number. Default: `1`
</ParamField>

<ParamField query="per_page" type="integer">
  Results per page. Default: `50`. Max: `200`
</ParamField>

<ParamField query="limit" type="integer">
  Alias for `per_page`
</ParamField>

<RequestExample>

```bash cURL
curl "https://api.northaxiumdata.ca/api/v1/politicians?party=Liberal&page=2&per_page=10" \
  -H "X-API-Key: <your-api-key>"
```

```python Python
import requests

url = "https://api.northaxiumdata.ca/api/v1/politicians"
headers = {"X-API-Key": "<your-api-key>"}
params = {
    "party": "Liberal",
    "page": 2,
    "per_page": 10
}

response = requests.get(url, headers=headers, params=params)
print(response.json())
```

```javascript Node.js
const fetch = require('node-fetch');

const url = new URL('https://api.northaxiumdata.ca/api/v1/politicians');
url.search = new URLSearchParams({
  party: 'Liberal',
  page: '2',
  per_page: '10'
});

fetch(url, {
  headers: {
    'X-API-Key': '<your-api-key>'
  }
})
  .then(res => res.json())
  .then(json => console.log(json));
```
